   419→      const n = topNOverride ?? topN;

   421→      const qs = new URLSearchParams({
   422→        topN: String(n),
   423→        rsiMin: String(f.rsiMin),
   424→        mcapMin: String(f.mcapMin),
   425→        volMin: String(f.volMin),
   426→      });
   427→      const endpoint = mode === "bridge" ? "/api/bridge-screen" : "/api/screen";
   428→      // Bridge screen takes ~70s — use AbortController with 120s timeout
   429→      const controller = new AbortController();
   430→      const timeoutId = setTimeout(() => controller.abort(), 120000);
   431→      const res = await fetch(`${endpoint}?${qs}`, {
   432→        cache: "no-store",
   433→        signal: controller.signal,
   434→      });
   435→      clearTimeout(timeoutId);
   436→      if (!res.ok) throw new Error(`Screen failed (${res.status})`);
   437→      const data: ScreenResponse = await res.json();

   439→      if (prev && prev.top10.length > 0 && data.top10.length > 0) {
   440→        const prevSyms = prev.top10.map((s) => s.symbol);
   441→        const currentSyms = data.top10.map((s) => s.symbol);
   442→        const added = currentSyms.filter((s) => !prevSyms.includes(s));
   443→        const removed = prevSyms.filter((s) => !currentSyms.includes(s));
   444→        if (added.length > 0 || removed.length > 0) {
   445→          const change: ScreenChange = {
   446→            date: new Date().toISOString(),
   447→            added: added.map((sym) => {
   448→              const stock = data.top10.find((s) => s.symbol === sym);
   449→              return { symbol: sym, ret6mPct: stock?.ret6mPct ?? 0 };
   450→            }),
   451→            removed,
   452→          };
   453→          setChangeHistory((prev) => [...prev, change].slice(-20));
   454→        }
   455→        setPrevScreenSymbols(prevSyms);
   456→      }
   457→      screenRef.current = data;
   458→      setScreen(data);
   459→      setAllocDirty(true);
   460→      if (!silent) {
   461→        if (data.error) {
   462→          toast({ title: "Screen failed — Yahoo rate-limited", description: data.error.slice(0, 100), variant: "destructive" });
   463→        } else if (data.warning) {
   464→          toast({ title: "Showing cached data", description: data.warning.slice(0, 100) });
   465→        } else {
   466→          const label = mode === "bridge" ? "Bridge screen" : "Screen";
   467→          toast({ title: `${label} complete`, description: `${data.top10.length} stocks in Top ${n}` });
   468→        }
   469→      }
   470→      return data;
   471→    } catch (e: any) {
   472→      const msg = e?.name === "AbortError"
   473→        ? "Screen timed out after 120s — Yahoo may be rate-limiting. Try again."
   474→        : e instanceof Error ? e.message : "Unknown";

   476→      return null;
   477→    } finally {
   478→      clearInterval(elapsedInterval);
   479→      setRunning(null);
   480→    }

   717→  // ===== Signal Score Smoothing + Stability + Consensus =====
   718→  // Computes for each stock in the current screen:
   719→  //   - smoothedScore: weighted average of last 3 runs (0.5 current + 0.3 prev + 0.2 prev-prev)
   720→  //   - stability: STABLE / VOLATILE / ERRATIC based on rank change across last 3 runs

   727→
   728→    for (const stock of screen.top10) {
   729→      const sym = stock.symbol;
   730→      const scores: number[] = [];
   731→      const ranks: number[] = [];
   732→      let appearances = 0;
   733→
   734→      for (const snap of recent) {
   735→        const found = snap.stocks.find((s: any) => s.symbol === sym);
   736→        if (found) {
   737→          scores.push(found.signalScore);
   738→          ranks.push(found.rank);

   754→        const rankChanges = ranks.slice(1).map((r, i) => Math.abs(r - ranks[i]));
   755→        const maxChange = Math.max(...rankChanges);
   756→        if (maxChange <= 2) stability = "STABLE";
   757→        else if (maxChange <= 5) stability = "VOLATILE";
   758→        else stability = "ERRATIC";
   759→      } else if (ranks.length === 1) {
   760→        stability = "NEW";
   761→      }
   762→
   763→      // Consensus: appeared in 2+ of last 3 runs
   764→      const consensus = appearances >= 2;
   765→
   766→      result[sym] = { smoothedScore, stability, consensus, appearances, prevRanks: ranks };
   767→    }
